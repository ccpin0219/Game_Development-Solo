// Lathed low-poly profiles create a genuinely hollow bowl and faceted stone base.
System.Func<string,Vector2[],Material,GameObject> lathe=(name,profile,m)=>{
    int n=12;var vertices=new Vector3[profile.Length*(n+1)];
    var triangles=new System.Collections.Generic.List<int>();
    for(int j=0;j<profile.Length;j++)for(int i=0;i<=n;i++){
        float a=i*Mathf.PI*2/n+Mathf.PI/12;
        vertices[j*(n+1)+i]=new Vector3(Mathf.Cos(a)*profile[j].x,profile[j].y,Mathf.Sin(a)*profile[j].x);
        if(j<profile.Length-1 && i<n){int k=j*(n+1)+i;triangles.AddRange(new[]{k,k+n+1,k+1,k+1,k+n+1,k+n+2});}
    }
    var mesh=new Mesh();mesh.name=name;mesh.vertices=vertices;mesh.triangles=triangles.ToArray();mesh.RecalculateNormals();mesh.RecalculateBounds();
    AssetDatabase.CreateAsset(mesh,folder+"/"+name+".asset");
    var g=new GameObject(name);g.transform.SetParent(template.transform,false);g.AddComponent<MeshFilter>().sharedMesh=mesh;g.AddComponent<MeshRenderer>().sharedMaterial=m;return g;
};

// Twisted tapering tongues give the flame a pointed silhouette, including in edit mode.
System.Action<string,Vector3,float,float,Material> tongue=(name,offset,height,width,m)=>{
    int rings=5,sides=7;var vertices=new Vector3[rings*sides];var triangles=new System.Collections.Generic.List<int>();
    float[] radii={.75f,1f,.72f,.36f,0f};
    for(int j=0;j<rings;j++)for(int i=0;i<sides;i++){
        float t=j/(float)(rings-1),a=i*Mathf.PI*2/sides+t*.9f;
        vertices[j*sides+i]=offset+new Vector3(Mathf.Cos(a)*width*radii[j]+Mathf.Sin(t*3)*width*.5f,t*height,Mathf.Sin(a)*width*radii[j]);
        if(j<rings-1){int k=j*sides+i,kn=j*sides+(i+1)%sides;triangles.AddRange(new[]{k,k+sides,kn,kn,k+sides,kn+sides});}
    }
    var mesh=new Mesh();mesh.name=name;mesh.vertices=vertices;mesh.triangles=triangles.ToArray();mesh.RecalculateNormals();mesh.RecalculateBounds();AssetDatabase.CreateAsset(mesh,folder+"/"+name+".asset");
    var g=new GameObject(name);g.transform.SetParent(flame.transform,false);g.AddComponent<MeshFilter>().sharedMesh=mesh;var r=g.AddComponent<MeshRenderer>();r.sharedMaterial=m;r.shadowCastingMode=UnityEngine.Rendering.ShadowCastingMode.Off;
};
